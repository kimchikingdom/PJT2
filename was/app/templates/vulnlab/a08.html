취약한 코드

import pickle, base64

# 신뢰하지 않는 입력을 직접 역직렬화
data = request.form.get("payload")
raw = base64.b64decode(data)
obj = pickle.loads(raw)  # 임의 코드 실행!

안전한 코드

import json

# JSON 사용: 코드 실행 불가
data = request.form.get("payload")
obj = json.loads(data)  # 안전
