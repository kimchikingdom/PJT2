취약한 코드

# Fail-Open: 예외 시 접근 허용
def check_permission(user):
    try:
        return user.role == "admin"
    except:
        return True  # 예외 → 허용!

# 예외 상세 노출
try:
    result = 1 / int(x)
except Exception as e:
    return str(e), 500  # traceback 노출

안전한 코드

# Fail-Closed: 예외 시 접근 거부
def check_permission(user):
    try:
        return user.role == "admin"
    except:
        return False  # 예외 → 거부!

# 안전한 예외 처리
try:
    result = 1 / int(x)
except ZeroDivisionError:
    abort(400, "0으로 나눌 수 없습니다")
except ValueError:
    abort(400, "숫자를 입력하세요")
