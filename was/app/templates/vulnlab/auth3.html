❌ 취약한 코드 (현재)

@app.route("/admin/user-list")
# @login_required  ← 누락!
# @admin_required  ← 누락!
def admin_user_list():
    users = User.query.all()
    return render_template("users.html",
                           users=users)

✅ 안전한 코드

@app.route("/admin/user-list")
@login_required   # ← 로그인 체크
@admin_required   # ← 관리자 권한 체크
def admin_user_list():
    users = User.query.all()
    return render_template("users.html",
                           users=users)
