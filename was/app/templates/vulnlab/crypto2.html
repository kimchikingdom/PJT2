❌ 취약한 코드

import hashlib

# MD5: Rainbow table로 즉시 복원
pw_hash = hashlib.md5(
    password.encode()
).hexdigest()

db.execute(
    "INSERT INTO users(username,password)"
    f" VALUES('{username}','{pw_hash}')"
)

✅ 안전한 코드

from werkzeug.security import generate_password_hash

# scrypt/PBKDF2 + 랜덤 salt 자동 적용
pw_hash = generate_password_hash(password)
# 결과: pbkdf2:sha256:600000$salt$hash

# 파라미터 바인딩으로 SQLi 차단
db.execute(
    "INSERT INTO users(username,password)"
    " VALUES(:u, :p)",
    {"u": username, "p": pw_hash}
)
